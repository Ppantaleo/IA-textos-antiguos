% Preámbulo común de los PDF para lectura e impresión
\usepackage{etoolbox}
\usepackage{booktabs}
\usepackage{longtable}
\usepackage{array}
\usepackage{microtype}
\usepackage{titlesec}
\usepackage{fancyhdr}
\usepackage{enumitem}
\usepackage[htt]{hyphenat}  % permite partir rutas en monoespaciada

% Tipografía
\setmainfont{TeX Gyre Pagella}
\setsansfont{TeX Gyre Heros}
\setmonofont{TeX Gyre Cursor}[Scale=MatchLowercase]

% Color de acento, discreto
\definecolor{acento}{HTML}{3A5A40}
\definecolor{gris}{HTML}{5A6B73}

% Títulos
\titleformat{\section}{\large\bfseries\color{acento}}{}{0pt}{}
\titleformat{\subsection}{\normalsize\bfseries}{}{0pt}{}
\titleformat{\subsubsection}{\normalsize\itshape}{}{0pt}{}
\titlespacing*{\section}{0pt}{2.2\baselineskip}{0.8\baselineskip}
\titlespacing*{\subsection}{0pt}{1.4\baselineskip}{0.5\baselineskip}

% Encabezado y pie
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0.4pt}
\renewcommand{\footrule}{\hbox to\headwidth{\color{gris}\leaders\hrule height \footrulewidth\hfill}}
\fancyfoot[L]{\footnotesize\color{gris}\documentoid}
\fancyfoot[R]{\footnotesize\color{gris}\thepage}

% Tablas: cuerpo menor y filas algo más aireadas
\AtBeginEnvironment{longtable}{\footnotesize}
\renewcommand{\arraystretch}{1.15}

% Citas en bloque: filete lateral verde
\usepackage{mdframed}
\newmdenv[linewidth=3pt, linecolor=acento, topline=false, bottomline=false,
          rightline=false, innerleftmargin=10pt, innerrightmargin=0pt,
          innertopmargin=6pt, innerbottommargin=6pt,
          skipabove=0.7\baselineskip, skipbelow=0.5\baselineskip]{cita}
\renewenvironment{quote}{\begin{cita}}{\end{cita}}

% Identificador del documento en el pie
\providecommand{\documentoid}{}

% Índice más compacto
\setcounter{tocdepth}{3}

% Evita desbordes cuando una ruta larga no admite corte
\setlength{\emergencystretch}{3em}
\sloppy  % prefiere una línea algo holgada a una ruta que se salga del margen

% La portada va sola; el índice empieza en página nueva
\pretocmd{\tableofcontents}{\clearpage}{}{}
